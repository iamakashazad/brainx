       ```python
       from __future__ import annotations

       from io import BytesIO
       from pathlib import Path

       import magic
       from PIL import Image, ImageOps

       ALLOWED_MIMES = frozenset({"image/png", "image/jpeg", "image/webp"})
       MAX_INPUT_BYTES = 5 * 1024 * 1024  # 5 MB hard cap (D-09)
       LONGEST_EDGE = 1024
       JPEG_QUALITY = 85
       MAX_DECODED_PIXELS = 64_000_000  # ~8000x8000 max — defense against decompression bombs (T-02-02-1)

       Image.MAX_IMAGE_PIXELS = MAX_DECODED_PIXELS


       class ImageValidationError(ValueError):
           """Raised when an upload fails MIME / size / decode checks."""


       def process_upload(raw: bytes, out_path: Path) -> dict:
           """
           Validate, sanitize, resize, and persist an uploaded image.

           Returns metadata: {width, height, bytes, mime}. Raises ImageValidationError
           on any policy violation. Synchronous (D-09) — caller is the request handler.
           """
           # 1. Size precheck — protects against decompression bombs BEFORE decode
           if len(raw) > MAX_INPUT_BYTES:
               raise ImageValidationError(
                   f"image > 5MB ({len(raw)} bytes) — max {MAX_INPUT_BYTES}"
               )
           if len(raw) == 0:
               raise ImageValidationError("empty upload")

           # 2. MIME sniff (libmagic) — defeats Content-Type spoofing
           detected = magic.from_buffer(raw[:2048], mime=True)
           if detected not in ALLOWED_MIMES:
               raise ImageValidationError(f"unsupported MIME {detected!r}")

           # 3. Decode + sanitize via Pillow
           try:
               with Image.open(BytesIO(raw)) as im:
                   # Apply EXIF orientation BEFORE stripping (Pitfall 5)
                   im = ImageOps.exif_transpose(im)
                   # Drop alpha / palette — JPEG cannot store alpha
                   im = im.convert("RGB")
                   # Resize so longest edge ≤ 1024 px (Lanczos for downscale)
                   im.thumbnail((LONGEST_EDGE, LONGEST_EDGE), Image.Resampling.LANCZOS)
                   # 4. Persist — UUID-only path (no user filename — mitigates path traversal T-02-02-1)
                   out_path.parent.mkdir(parents=True, exist_ok=True)
                   # No exif= kwarg → EXIF is stripped (T-02-02-3)
                   im.save(out_path, format="JPEG", quality=JPEG_QUALITY, optimize=True)
                   return {
                       "width": im.width,
                       "height": im.height,
                       "bytes": out_path.stat().st_size,
                       "mime": detected,
                   }
           except Image.DecompressionBombError as exc:
               raise ImageValidationError(f"decompression bomb refused: {exc}") from exc
           except (OSError, ValueError, Image.UnidentifiedImageError) as exc:
               raise ImageValidationError(f"image decode failed: {exc}") from exc
       ```

       ```python
       from __future__ import annotations

       import uuid

       from fastapi import APIRouter, File, HTTPException, UploadFile, status
       from sqlalchemy import select

       from backend.core import database
       from backend.core.config import settings
       from backend.models.content import Content, ContentType
       from backend.models.schemas import ContentCreated, TextContentRequest
       from backend.services.content_processing import (
           ImageValidationError,
           process_upload,
       )

       router = APIRouter()


       @router.post("/text", response_model=ContentCreated, status_code=status.HTTP_201_CREATED)
       async def create_text_content(payload: TextContentRequest) -> ContentCreated:
           content_id = str(uuid.uuid4())
           row = Content(
               id=content_id,
               content_type=ContentType.TEXT.value,
               content_text=payload.content_text,
               screen_count=None,
               meta_json=None,
           )
           async with database.async_session() as session:
               session.add(row)
               await session.commit()
           return ContentCreated(content_id=content_id, content_type=ContentType.TEXT.value)


       @router.post(
           "/screens",
           response_model=ContentCreated,
           status_code=status.HTTP_201_CREATED,
       )
       async def create_ui_screens(files: list[UploadFile] = File(...)) -> ContentCreated:
           if not 1 <= len(files) <= 5:
               raise HTTPException(status_code=400, detail="ui_screens requires 1–5 images")

           content_id = str(uuid.uuid4())
           target_dir = settings.uploads_dir / content_id
           per_file_meta = []
           for index, upload in enumerate(files, start=1):
               raw = await upload.read()  # read ONCE (Pitfall 6)
               try:
                   meta = process_upload(raw, target_dir / f"screen_{index}.jpg")
               except ImageValidationError as exc:
                   raise HTTPException(status_code=400, detail=str(exc)) from exc
               per_file_meta.append({**meta, "original_filename": upload.filename, "index": index})

           row = Content(
               id=content_id,
               content_type=ContentType.UI_SCREENS.value,
               screen_count=len(files),
               content_text=None,
               meta_json={"files": per_file_meta},
           )
           async with database.async_session() as session:
               session.add(row)
               await session.commit()
           return ContentCreated(content_id=content_id, content_type=ContentType.UI_SCREENS.value)


       @router.post(
           "/static",
           response_model=ContentCreated,
           status_code=status.HTTP_201_CREATED,
       )
       async def create_static_content(file: UploadFile = File(...)) -> ContentCreated:
           content_id = str(uuid.uuid4())
           target_dir = settings.uploads_dir / content_id
           raw = await file.read()
           try:
               meta = process_upload(raw, target_dir / "image.jpg")
           except ImageValidationError as exc:
               raise HTTPException(status_code=400, detail=str(exc)) from exc
           row = Content(
               id=content_id,
               content_type=ContentType.STATIC.value,
               screen_count=1,
               content_text=None,
               meta_json={"files": [{**meta, "original_filename": file.filename, "index": 1}]},
           )
           async with database.async_session() as session:
               session.add(row)
               await session.commit()
           return ContentCreated(content_id=content_id, content_type=ContentType.STATIC.value)
       ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP multipart → /content/{screens,static} | Untrusted bytes; treated as adversarial until validated through process_upload |
| HTTP JSON → /content/text | Untrusted string; bounded by Pydantic max_length |
| Filesystem write under uploads_dir | Server writes; path is uuid-only — no user input in path components |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-02-1 | Tampering / Spoofing | MIME spoof — `.exe` bytes labeled `.jpg`; path traversal via filename like `../../etc/passwd.jpg`; decompression bombs | mitigate | Task 1: `magic.from_buffer(raw[:2048])` validates real MIME; `Image.MAX_IMAGE_PIXELS` caps decoded pixels; `ImageValidationError` raised pre-decode for size; **filenames never enter the path** — only `uploads/{uuid}/screen_{n}.jpg` literal pattern. Tested via test_mime_spoofing_rejected and test_oversize_rejected |
| T-02-02-2 | Denial of Service | Disk exhaustion — many uploads of 5MB images | mitigate | 5MB per-file cap + 5-image-per-flow cap (D-09/D-10) bound a single request to 25MB. **Volume-rate-limiting deferred to Phase 5 (queue-depth 429)**. Acceptable v1 risk — single-user research preview |
