       ```python
       from __future__ import annotations

       from typing import Literal

       from pydantic import BaseModel, Field

       from backend.models.simulation import Event, SimulationLog


       class HesitationPoint(BaseModel):
           x_ratio: float = Field(ge=0.0, le=1.0)
           y_ratio: float = Field(ge=0.0, le=1.0)
           duration_ms: int = Field(ge=0)
           intensity: float = Field(ge=0.0, le=1.0)


       class ScrollPoint(BaseModel):
           t_ms: int = Field(ge=0)
           y_ratio: float = Field(ge=0.0, le=1.0)


       class ScreenOverlay(BaseModel):
           screen_index: int
           hesitation_points: list[HesitationPoint] = Field(default_factory=list)
           scroll_path: list[ScrollPoint] | None = None


       CLUSTER_RADIUS = 0.1  # ratio space (per RESEARCH.md §Overlay derivation)
       DEFAULT_HESITATION_DURATION_MS = 300
       INTENSITY_NORMALIZER = 3.0  # divisor for summed intensity (per algorithm sketch step 3)


       def _cluster_hesitations(events: list[Event]) -> list[HesitationPoint]:
           """
           Greedy clustering in (x_ratio, y_ratio) space, radius CLUSTER_RADIUS.
           Deterministic: events processed in order of (-intensity, timestamp_ms, x_ratio, y_ratio)
           to break ties stably.
           """
           candidates = [
               (e.intensity, e.timestamp_ms, e.target_region.x_ratio, e.target_region.y_ratio, e)
               for e in events
               if e.type == "hesitation" and e.target_region is not None
           ]
           # Stable, deterministic sort: highest intensity first, then earliest timestamp
           candidates.sort(key=lambda c: (-c[0], c[1], c[2], c[3]))

           clusters: list[dict] = []
           for intensity, _ts, x, y, _ev in candidates:
               assigned = False
               for cluster in clusters:
                   if (cluster["x"] - x) ** 2 + (cluster["y"] - y) ** 2 <= CLUSTER_RADIUS ** 2:
                       # Weighted-average centroid by accumulated intensity
                       new_total = cluster["intensity_sum"] + intensity
                       cluster["x"] = (cluster["x"] * cluster["intensity_sum"] + x * intensity) / new_total
                       cluster["y"] = (cluster["y"] * cluster["intensity_sum"] + y * intensity) / new_total
                       cluster["intensity_sum"] = new_total
                       cluster["count"] += 1
                       assigned = True
                       break
               if not assigned:
                   clusters.append({"x": x, "y": y, "intensity_sum": intensity, "count": 1})

           return [
               HesitationPoint(
                   x_ratio=round(c["x"], 6),
                   y_ratio=round(c["y"], 6),
                   duration_ms=c["count"] * DEFAULT_HESITATION_DURATION_MS,
                   intensity=min(1.0, c["intensity_sum"] / INTENSITY_NORMALIZER),
               )
               for c in clusters
           ]


       def _build_scroll_path(events: list[Event]) -> list[ScrollPoint]:
           scroll = [
               ScrollPoint(t_ms=e.timestamp_ms, y_ratio=e.target_region.y_ratio)
               for e in events
               if e.type == "scroll" and e.target_region is not None
           ]
           scroll.sort(key=lambda p: p.t_ms)
           return scroll


       def derive_overlays(
           sim_log: SimulationLog,
           content_type: Literal["ui_screens", "static", "text"] = "ui_screens",
       ) -> list[ScreenOverlay]:
           """
           Deterministic overlay derivation from sim_log.events (D-15).
           For text/static content, scroll_path is forced to None regardless of events.
           """
           overlays: list[ScreenOverlay] = []
           for screen in sim_log.screens:
               hesitation_points = _cluster_hesitations(screen.events)
               if content_type in ("text", "static"):
                   scroll_path = None
               else:
                   scroll_path_list = _build_scroll_path(screen.events)
                   scroll_path = scroll_path_list if scroll_path_list else None
               overlays.append(
                   ScreenOverlay(
                       screen_index=screen.screen_index,
                       hesitation_points=hesitation_points,
                       scroll_path=scroll_path,
                   )
               )
           overlays.sort(key=lambda o: o.screen_index)
           return overlays
       ```

       ```python
       from __future__ import annotations

       import asyncio
       import logging
       from pathlib import Path

       import numpy as np
       from sqlalchemy import select
       from sqlalchemy.ext.asyncio import AsyncSession

       from backend.core import database
       from backend.models.content import Content
       from backend.models.job import Job, StageEnum
       from backend.models.persona import Persona
       from backend.models.simulation import (
           HallucinationError,
           SchemaValidationFailure,
       )
       from backend.services import grounding, simulation
       from backend.services.inference import run_tribev2_text, run_tribev2_video
       from backend.services.overlay import derive_overlays

       logger = logging.getLogger(__name__)

       gpu_queue: asyncio.Queue[str] = asyncio.Queue()


       def _pending_job_ids() -> list[str]:
           return list(gpu_queue._queue)  # noqa: SLF001 — queue order required for position tracking


       async def sync_pending_queue_positions() -> None:
           pending_ids = _pending_job_ids()
           async with database.async_session() as session:
               result = await session.execute(select(Job).where(Job.status == "PENDING"))
               jobs = {job.id: job for job in result.scalars()}
               for job in jobs.values():
                   job.queue_position = None
               for index, job_id in enumerate(pending_ids, start=1):
                   job = jobs.get(job_id)
                   if job is not None:
                       job.queue_position = index
               await session.commit()


       async def transition(session: AsyncSession, job: Job, stage: StageEnum | None) -> None:
           """Centralized stage write — D-19. All stage mutations MUST go through here."""
           job.stage = stage.value if stage is not None else None
           await session.commit()
           logger.info("job %s → %s", job.id, job.stage)


       async def gpu_worker_loop() -> None:
           while True:
               job_id = await gpu_queue.get()
               try:
                   await _process_job(job_id)
               finally:
                   gpu_queue.task_done()


       async def _process_job(job_id: str) -> None:
           # Load job + linked persona + content (if any)
           async with database.async_session() as session:
               result = await session.execute(select(Job).where(Job.id == job_id))
               job = result.scalar_one_or_none()
               if job is None:
                   return
               job.status = "RUNNING"
               job.queue_position = None
               persona: Persona | None = None
               content: Content | None = None
               if job.persona_id:
                   pres = await session.execute(select(Persona).where(Persona.id == job.persona_id))
                   persona = pres.scalar_one_or_none()
               if job.content_id:
                   cres = await session.execute(select(Content).where(Content.id == job.content_id))
                   content = cres.scalar_one_or_none()
               # Phase 1 path: no persona AND no content → legacy /test-inference job
               legacy_phase1 = persona is None and content is None
               input_payload = job.input_path or ""
               input_type = job.input_type or "text"
               await session.commit()

           await sync_pending_queue_positions()

           # ============== LEGACY PHASE-1 PATH (D-12 non-breaking) ==============
           if legacy_phase1:
               await _run_inference_only(job_id, input_type, input_payload)
               return

           # ============== PHASE 2 5-STAGE LIFECYCLE (D-18..D-20) ==============
           try:
               # ---- SIMULATION (D-20: simulation.simulate handles its own repair-once) ----
               async with database.async_session() as session:
                   job = (await session.execute(select(Job).where(Job.id == job_id))).scalar_one()
                   await transition(session, job, StageEnum.SIMULATION)
               sim_log = await simulation.simulate(persona, content)

               # ---- GROUNDING_CHECK (D-20: no retry on hallucination) ----
               async with database.async_session() as session:
                   job = (await session.execute(select(Job).where(Job.id == job_id))).scalar_one()
                   await transition(session, job, StageEnum.GROUNDING_CHECK)
               await grounding.judge(content, sim_log)  # raises HallucinationError on flag

               # ---- INFERENCE (D-20: inherits Phase 1 D-07 single-retry) ----
               async with database.async_session() as session:
                   job = (await session.execute(select(Job).where(Job.id == job_id))).scalar_one()
                   await transition(session, job, StageEnum.INFERENCE)
               # Build TRIBEv2 input from sim_log narrative (text path only — DEFAULT_PATHWAY=text post-01-04)
               narrative = "\n\n".join(s.narrative_text for s in sim_log.screens)
               activation = await _run_inference_with_retry(narrative)

               # ---- PROCESSING (D-20: pure code, fail-hard) ----
               async with database.async_session() as session:
                   job = (await session.execute(select(Job).where(Job.id == job_id))).scalar_one()
                   await transition(session, job, StageEnum.PROCESSING)
               overlays = derive_overlays(sim_log, content_type=content.content_type)
               Path("activations").mkdir(exist_ok=True)
               output_path = Path("activations") / f"{job_id}.npy"
               np.save(output_path, activation)

               # ---- DONE — write result_json by REASSIGNMENT (Pitfall 4) ----
               async with database.async_session() as session:
                   job = (await session.execute(select(Job).where(Job.id == job_id))).scalar_one()
                   job.result_json = {
                       **(job.result_json or {}),
                       "simulation": sim_log.model_dump(),
                       "overlays": [o.model_dump() for o in overlays],
                       "activation": {"path": output_path.as_posix(), "shape": list(activation.shape)},
                   }
                   job.status = "DONE"
                   await transition(session, job, None)
                   await session.commit()
           except HallucinationError as exc:
               async with database.async_session() as session:
                   job = (await session.execute(select(Job).where(Job.id == job_id))).scalar_one()
                   job.status = "ERROR"
                   job.error_message = str(exc)
       ```

       ```python
       from backend.models.content import Content
       from backend.models.job import Job
       from backend.models.persona import Persona
       from backend.models.schemas import AnalyzeRequest, AnalyzeResponse

       router = APIRouter()


       @router.post("", response_model=AnalyzeResponse, status_code=status.HTTP_201_CREATED)
       async def submit_analyze(payload: AnalyzeRequest) -> AnalyzeResponse:
           """
           D-11: Validate FKs, create Job row, enqueue, return PENDING immediately.
           No long-running work in this handler — all DeepSeek/TRIBEv2 calls live in the worker.
           """
           async with database.async_session() as session:
               persona = (await session.execute(select(Persona).where(Persona.id == payload.persona_id))).scalar_one_or_none()
               if persona is None:
                   raise HTTPException(status_code=404, detail=f"Persona {payload.persona_id} not found")
               content = (await session.execute(select(Content).where(Content.id == payload.content_id))).scalar_one_or_none()
               if content is None:
                   raise HTTPException(status_code=404, detail=f"Content {payload.content_id} not found")

               job = Job(
                   id=str(uuid.uuid4()),
                   status="PENDING",
                   queue_position=queue_manager.gpu_queue.qsize() + 1,
                   persona_id=payload.persona_id,
                   content_id=payload.content_id,
               )
               session.add(job)
               await session.commit()

           await queue_manager.gpu_queue.put(job.id)
           await queue_manager.sync_pending_queue_positions()
           return AnalyzeResponse(job_id=job.id, status="PENDING")
       ```
