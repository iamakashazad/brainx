### Step A: `backend/routers/activations.py`
New FastAPI router. Endpoint `GET /jobs/{job_id}/activations/{screen_index}`.

```python
from __future__ import annotations
from pathlib import Path
import numpy as np
from fastapi import APIRouter, HTTPException, Response
from sqlalchemy import select
from backend.core import database
from backend.models.job import Job
from backend.models.results import AnalysisResult

router = APIRouter()

VERTEX_COUNT = 20484  # fsaverage5 bilateral; verified in backend/services/roi_aggregation.py

@router.get("/jobs/{job_id}/activations/{screen_index}")
async def get_screen_activation(job_id: str, screen_index: int) -> Response:
    async with database.async_session() as session:
        result = await session.execute(select(Job).where(Job.id == job_id))
        job = result.scalar_one_or_none()

    if job is None or job.status != "DONE" or job.result_json is None:
        raise HTTPException(status_code=404, detail=f"Job {job_id} not found or not complete")

    envelope = AnalysisResult.model_validate(job.result_json)
    manifest_entry = next(
        (entry for entry in envelope.activations if entry.screen_index == screen_index),
        None,
    )
    if manifest_entry is None:
        raise HTTPException(status_code=404, detail=f"Screen {screen_index} not found in job {job_id}")

    path = Path(manifest_entry.path)
    if not path.is_absolute():
        path = Path.cwd() / path
    if not path.exists():
        raise HTTPException(status_code=404, detail=f"Activation file missing on disk: {manifest_entry.path}")

    arr = np.load(path)  # shape (T, 20484) per backend/core/queue.py
    if arr.ndim != 2 or arr.shape[1] != VERTEX_COUNT:
        raise HTTPException(
            status_code=500,
            detail=f"Unexpected activation shape {arr.shape} — expected (T, {VERTEX_COUNT})",
        )

    time_mean = arr.mean(axis=0).astype(np.float32)
    body = time_mean.tobytes(order="C")  # little-endian on x86/ARM64

    return Response(
        content=body,
        media_type="application/octet-stream",
        headers={
            "X-Vertex-Count": str(VERTEX_COUNT),
            "X-Shape": f"{VERTEX_COUNT}",
            "Cache-Control": "public, max-age=300",  # activations are immutable once DONE
        },
    )
```

### Step D: `scripts/bake_fsaverage5.py`
One-off Python script. Reads fsaverage5 pial mesh via `nilearn.datasets.fetch_surf_fsaverage()` (already cached under `model_cache/nilearn/` per `backend/services/roi_aggregation.py:64`) and the Destrieux atlas via `fetch_atlas_surf_destrieux`. Outputs two files under `frontend/public/brain/`:

1. `fsaverage5_pial_bilateral.bin` — a single binary blob:
   - 8-byte header: `uint32 vertex_count`, `uint32 triangle_count`.
   - Then `vertex_count * 3 * float32` positions (little-endian) — left hemisphere first, then right (offset right triangle indices by `left_vertex_count`).
   - Then `triangle_count * 3 * uint32` triangle indices.
2. `fsaverage5_parcellation.json` — `{ "labels": string[], "circuit_of_label_index": (string | null)[], "vertex_to_label_index": number[] }`. The `circuit_of_label_index` array uses the `ROI_CIRCUIT_MAP` from `backend/services/roi_aggregation.py` — any atlas label appearing in one of the five circuit lists gets that circuit name; all others get `null`.

Run script once during plan execution: `python scripts/bake_fsaverage5.py`. The resulting `.bin` and `.json` files ARE committed to git (they are ~300 KB gzipped).

```ts
export function useActivation(jobId: string | null, screenIndex: number) {
  return useQuery<Float32Array>({
    queryKey: ["activation", jobId, screenIndex],

    queryFn: async ({ signal }) => {
      const res = await fetch(
        `${import.meta.env.VITE_API_BASE_URL ?? "http://localhost:8000"}/jobs/${jobId}/activations/${screenIndex}`,
        { signal }
      );
      if (!res.ok) throw new Error(`Activation fetch failed: ${res.status}`);
      const buf = await res.arrayBuffer();
      const arr = new Float32Array(buf);
      if (arr.length !== 20484) {
        throw new Error(`Unexpected activation length ${arr.length} — expected 20484`);
      }
      return arr;
    },
```

### Step F: `frontend/src/features/results/colorBuffer.ts` (PURE FUNCTION — canvas-free; fully unit-testable)
```ts
import type { CircuitName } from "@/lib/api-types";
import { NEUTRAL_BASE, getTokenColor } from "./oklchToLinear";

export const CIRCUIT_CHART_TOKEN: Record<CircuitName, string> = {
  "Attention": "--chart-1",
  "Cognitive Load": "--chart-2",
  "Emotional Response": "--chart-3",
  "Memory Encoding": "--chart-4",
  "Reward": "--chart-5",
};

export function computeColorBuffer(opts: {
  activations: Float32Array;            // length 20484
  circuitOfLabelIndex: (string | null)[];
  vertexToLabelIndex: number[];
  activeCircuit: CircuitName;
  peakColor?: [number, number, number]; // override for tests
  baseColor?: [number, number, number];
}): Float32Array {
  const { activations, circuitOfLabelIndex, vertexToLabelIndex, activeCircuit } = opts;
  if (activations.length !== vertexToLabelIndex.length) {
    throw new Error(`Activation length ${activations.length} does not match vertex count ${vertexToLabelIndex.length}`);
  }
  const base = opts.baseColor ?? NEUTRAL_BASE;
  const peak = opts.peakColor ?? getTokenColor(CIRCUIT_CHART_TOKEN[activeCircuit]);

  // Isolate active-circuit activations for 20th-percentile threshold
  const inCircuit: number[] = [];
  for (let v = 0; v < activations.length; v++) {
    const label = vertexToLabelIndex[v];
    if (circuitOfLabelIndex[label] === activeCircuit) inCircuit.push(activations[v]);
  }
  const sorted = [...inCircuit].sort((a, b) => a - b);
  const p20 = sorted.length > 0 ? sorted[Math.floor(sorted.length * 0.2)] : 0;
  const minA = p20;
  const maxA = sorted.length > 0 ? sorted[sorted.length - 1] : 1;
  const range = Math.max(maxA - minA, 1e-6);

  const out = new Float32Array(activations.length * 3);
  for (let v = 0; v < activations.length; v++) {
    const label = vertexToLabelIndex[v];
    const inA = circuitOfLabelIndex[label] === activeCircuit;
    const a = activations[v];
    const below = a < p20;
    if (!inA || below) {
      out[v * 3] = base[0]; out[v * 3 + 1] = base[1]; out[v * 3 + 2] = base[2];
    } else {
      const t = Math.min(1, Math.max(0, (a - minA) / range));
      out[v * 3]     = base[0] + (peak[0] - base[0]) * t;
      out[v * 3 + 1] = base[1] + (peak[1] - base[1]) * t;
      out[v * 3 + 2] = base[2] + (peak[2] - base[2]) * t;
    }
  }
  return out;
}
```

```ts
import type { Recommendation } from "@/lib/api-types";

export type RecWithId = Recommendation & { id: string };

export function withStableIds(recs: Recommendation[]): RecWithId[] {
  return recs.map((r, idx) => ({
    ...r,
    id: `${r.screen_index}-${r.rec_type}-${idx}`,
  }));
}

export function groupRecsByScreen(recs: RecWithId[]): Record<number, RecWithId[]> {
  const grouped: Record<number, RecWithId[]> = {};
  for (const r of recs) {
    const k = r.screen_index;
    grouped[k] = grouped[k] ?? [];
    grouped[k].push(r);
  }
  return grouped;
}

/**
 * For D-10 "cycle through secondary locations" — within a screen group,
 * advance from the active rec to the next one (wrapping).
 */
export function cycleToSecondary(
  activeRecId: string,
  cursor: number,
  sameScreenGroup: RecWithId[]
): RecWithId {
  if (sameScreenGroup.length <= 1) return sameScreenGroup[0]; // no-op
  const activeIdx = sameScreenGroup.findIndex((r) => r.id === activeRecId);
  const nextIdx = (activeIdx + 1 + cursor) % sameScreenGroup.length;
  return sameScreenGroup[nextIdx];
}
```
