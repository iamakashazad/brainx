```ts
// GET /personas → PersonaRead[]
// POST /personas → PersonaRead (201)
// GET /personas/{id}, PATCH /personas/{id}, DELETE /personas/{id}
type PersonaRead = {
  id: string; name: string;
  demographics: { age: number; role: string; background: string | null };
  psychographics: { goals: string[]; values: string[]; motivations: string[] };
  cognitive_state: {
    attention_span: "short" | "medium" | "long";
    stress_level: "low" | "moderate" | "high";
    tech_literacy: "novice" | "intermediate" | "expert";
  };
  environment: {
    device: "mobile" | "tablet" | "desktop";
    location: "commute" | "home" | "office" | "store" | "other";
    ambient: string | null;
  };
  created_at: string; updated_at: string;
};
type PersonaCreate = Omit<PersonaRead, "id" | "created_at" | "updated_at">;

// POST /content/text  body: { content_text: string }  (1..20000 chars) → { content_id: string; content_type: "text" }
// POST /content/screens  multipart: files: File[] (1..5) → { content_id; content_type: "ui_screens" }
// POST /content/static   multipart: file: File       → { content_id; content_type: "static" }
type ContentCreated = { content_id: string; content_type: "text" | "ui_screens" | "static" };

// POST /analyze  body: { persona_id: string; content_id: string } → { job_id: string; status: "PENDING" }
type AnalyzeResponse = { job_id: string; status: "PENDING" };
```

### Step 13: `frontend/src/lib/api.ts`

```ts
import { persist } from "zustand/middleware";

type WizardState = {
  currentStep: 1 | 2 | 3 | 4;
  selectedPersonaId: string | null;
  contentId: string | null;
  contentType: "text" | "ui_screens" | "static" | null;

  activeCircuit: CircuitName | null;
  activeRecId: string | null;
  secondaryRecLocationCursor: number;

  setStep: (s: 1 | 2 | 3 | 4) => void;
  setPersona: (id: string | null) => void;
  setContent: (id: string, type: "text" | "ui_screens" | "static") => void;
  setJob: (id: string | null) => void;
  setActiveScreenIndex: (i: number) => void;
  setActiveCircuit: (c: CircuitName | null) => void;
  setActiveRecId: (id: string | null) => void;
  bumpSecondaryCursor: () => void;
  reset: () => void;
};

  persist(
    (set) => ({
      currentStep: 1,
      selectedPersonaId: null, contentId: null, contentType: null,
      jobId: null,
      activeScreenIndex: 0, activeCircuit: null, activeRecId: null,
      secondaryRecLocationCursor: 0,
      setStep: (s) => set({ currentStep: s }),
      setPersona: (id) => set({ selectedPersonaId: id }),
      setContent: (id, type) => set({ contentId: id, contentType: type }),
      setJob: (id) => set({ jobId: id }),
      setActiveScreenIndex: (i) => set({ activeScreenIndex: i }),
      setActiveCircuit: (c) => set({ activeCircuit: c }),
      setActiveRecId: (id) => set({ activeRecId: id, secondaryRecLocationCursor: 0 }),
      bumpSecondaryCursor: () => set((s) => ({ secondaryRecLocationCursor: s.secondaryRecLocationCursor + 1 })),
      reset: () => set({
        currentStep: 1, selectedPersonaId: null, contentId: null, contentType: null, jobId: null,
        activeScreenIndex: 0, activeCircuit: null, activeRecId: null, secondaryRecLocationCursor: 0,
      }),
    }),
    {
      name: "brainx-wizard-state",
      partialize: (s) => ({
        currentStep: s.currentStep,
        selectedPersonaId: s.selectedPersonaId,
        contentId: s.contentId,
        contentType: s.contentType,
        jobId: s.jobId,
      }),
    }
  )
);
```
