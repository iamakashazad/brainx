```ts
import type { JobStatus, StageValue } from "./api-types";

/**
 * Display copy — VERBATIM from UI-SPEC §Copywriting Contract, lines 246–250.
 * Do not paraphrase.
 */
export const DISPLAY_QUEUED = "Queued";
export const DISPLAY_SIMULATING =
  "Simulating how your persona experiences this content…";
export const DISPLAY_INFERRING = "Running brain inference…";
export const DISPLAY_AGGREGATING = "Aggregating results…";
export const DISPLAY_COMPLETE = "Analysis complete";
export const DISPLAY_FAILED_HEADING = "Analysis failed";
export const DISPLAY_FAILED_BODY =
  "Something went wrong during brain inference. Check that your content uploaded correctly, then try again.";

/**
 * Ordered list of the 4 display stages shown by StageIndicator.
 * Does NOT include the terminal "Analysis complete" label.
 */
export const stageOrder = [
  DISPLAY_QUEUED,
  DISPLAY_SIMULATING,
  DISPLAY_INFERRING,
  DISPLAY_AGGREGATING,
] as const;
export type DisplayStage = (typeof stageOrder)[number];

/**
 * Map (backend stage, backend status) → UI display copy string.
 *
 * Mapping (resolved per RESEARCH §Open Questions #1 and #4, VERBATIM):
 *   backend.stage           → display.stage.copy
 *   null  (status=PENDING)  → "Queued"
 *   "simulation"            → DISPLAY_SIMULATING
 *   "grounding_check"       → DISPLAY_SIMULATING    (folded)
 *   "inference"             → DISPLAY_INFERRING
 *   "roi_aggregation"       → DISPLAY_INFERRING     (folded)
 *   "processing"            → DISPLAY_AGGREGATING
 *   "recommendations"       → DISPLAY_AGGREGATING   (folded)
 *   null  (status=DONE)     → DISPLAY_COMPLETE
 *
 * Status-level helpers:
 *   backend "DONE"  → display "COMPLETE"
 *   backend "ERROR" → display "FAILED"
 */
export function displayStage(
  stage: StageValue | null | undefined,
  status: JobStatus
): string {
  if (status === "DONE") return DISPLAY_COMPLETE;
  if (status === "PENDING" || stage === null || stage === undefined) {
    return DISPLAY_QUEUED;
  }
  switch (stage) {
    case "simulation":
    case "grounding_check":
      return DISPLAY_SIMULATING;
    case "inference":
    case "roi_aggregation":
      return DISPLAY_INFERRING;
    case "processing":
    case "recommendations":
      return DISPLAY_AGGREGATING;
    default:
      return DISPLAY_QUEUED;
  }
}

/**
 * Index into `stageOrder` for the currently-active display stage.
 * Used by StageIndicator to highlight which stage bubble is active.
 */
export function displayStageIndex(
  stage: StageValue | null | undefined,
  status: JobStatus
): number {
  const label = displayStage(stage, status);
  const i = stageOrder.indexOf(label as DisplayStage);
  return i >= 0 ? i : 0;
}

/**
 * Display-only status label.
 * NOTE: actual JobStatus values in code are "DONE"/"ERROR"/"PENDING"/"RUNNING".
 * This is ONLY for text rendering to match UI-SPEC wording.
 */
export function displayStatusForStage(status: JobStatus): string {
  if (status === "DONE") return "COMPLETE";
  if (status === "ERROR") return "FAILED";
  if (status === "PENDING") return "PENDING";
  return "RUNNING";
}
```

### `frontend/src/features/run/useJobStatus.ts`
```ts
import { useQuery } from "@tanstack/react-query";
import { getJobStatus } from "@/lib/api";
import type { JobStatusResponse } from "@/lib/api-types";

export function useJobStatus(jobId: string | null) {
  return useQuery<JobStatusResponse>({
    queryKey: ["job", jobId],
    enabled: !!jobId,
    queryFn: ({ signal }) => getJobStatus(jobId!, signal),
    refetchInterval: (query) => {
      const status = query.state.data?.status;
      if (status === "DONE" || status === "ERROR") return false;
      return 2000;
    },
    refetchIntervalInBackground: false,
    // Fixed 5s retry cadence honors UI-SPEC line 260 ("auto-retry every 5s").
    // Do NOT use TanStack Query default exponential backoff — UI copy promises a constant cadence.
    retry: (failureCount) => failureCount < 3,
    retryDelay: 5000,
  });
}
```

### `frontend/src/features/run/useJobRehydrate.ts`
On mount, if `useWizardStore().jobId` is present, kicks off ONE `queryClient.fetchQuery({ queryKey: ["job", jobId], queryFn: …, staleTime: 0 })` to decide the initial state. Based on response:
- `DONE` → `setStep(4)`, do not start polling.
- `ERROR` → stay on step 3, let `ProgressView` render the failure state.
- `PENDING` or `RUNNING` → nothing (the normal `useJobStatus` observer in `ProgressView` picks up polling).
Returns `{ isRehydrating: boolean }` so `ProgressView` can show a subtle loading state for the first ~100ms.
