                   "circuit": circuit,

                   "peak":  float(peak_per_vertex[mask].max()),
               })

           for name, agg in circuits.items():
               n = per_circuit_count.get(name, 0)
               if n:
                   agg["score"] = agg["score"] / n

           return {"circuits": circuits, "regions": region_rows}

       def rescale_to_0_100(per_screen_bundles: list[dict]) -> list[dict]:
           """Global 0..100 rescale across all screens so per-screen scores remain comparable
           for BRAIN-03 continuity (Open Question 4). Applied at result_json write time per D-14.

           Input bundles use native TRIBEv2 scalars. Output bundles use 0..100 floats.
           """
           if not per_screen_bundles:
               return []
           # Collect all native scores (circuits.score, circuits.peak, regions.score, regions.peak)
           all_scores: list[float] = []
           for bundle in per_screen_bundles:
               for circ in bundle["circuits"].values():
                   all_scores.extend([circ["score"], circ["peak"]])
               for row in bundle["regions"]:
                   all_scores.extend([row["score"], row["peak"]])
           if not all_scores:
               return per_screen_bundles
           lo, hi = min(all_scores), max(all_scores)
           span = hi - lo if hi > lo else 1.0
           def _scale(x: float) -> float:
               return float(max(0.0, min(100.0, 100.0 * (x - lo) / span)))
           out: list[dict] = []
           for bundle in per_screen_bundles:
               out.append({
                   "circuits": {name: {"score": _scale(c["score"]), "peak": _scale(c["peak"])}
                                for name, c in bundle["circuits"].items()},
                   "regions":  [{"atlas_label": r["atlas_label"], "circuit": r["circuit"],
                                 "score": _scale(r["score"]), "peak": _scale(r["peak"])}
                                for r in bundle["regions"]],
               })
           return out
