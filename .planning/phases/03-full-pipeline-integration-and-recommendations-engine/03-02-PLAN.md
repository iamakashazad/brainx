       """Flow-level continuity Pydantic models (BRAIN-03, D-11).

       Per-circuit per-screen score_0_100 + delta_from_prev + overall trend tag.
       Computed deterministically in backend/services/continuity.py — no LLM involvement.
       """
       from __future__ import annotations

       from typing import Literal

       from pydantic import BaseModel, Field

       TrendTag = Literal["rising", "dropping", "flat", "volatile"]


       class ContinuityPoint(BaseModel):
           screen_index: int = Field(ge=0)
           score_0_100: float = Field(ge=0.0, le=100.0)
           delta_from_prev: float  # -100..100 in practice; no hard bound needed


       class CircuitContinuity(BaseModel):
           per_screen: list[ContinuityPoint]
           trend: TrendTag


       class FlowContinuity(BaseModel):
           """Flow-level continuity for all 5 named circuits (D-11)."""
           circuits: dict[str, CircuitContinuity]
       ```

    """Deterministic flow continuity service (BRAIN-03, D-11, D-12).

    Computes per-circuit per-screen score_0_100 + delta_from_prev + overall trend tag
    from rescaled per-screen ROI bundles. No LLM involvement.

    Thresholds are module-level constants so they can be tuned without code search (D-12).
    """
    from __future__ import annotations

    from typing import Literal

    from backend.models.continuity import CircuitContinuity, ContinuityPoint, FlowContinuity
    from backend.services.roi_aggregation import CIRCUIT_NAMES

    # Module-level constants — tunable without code search (D-12)
    DELTA_FLOOR: float = 5.0            # |delta| below this is treated as "flat"
    VOLATILITY_FLIPS_MIN: int = 2        # >= 2 sign-changes in significant deltas → volatile

    TrendTag = Literal["rising", "dropping", "flat", "volatile"]


    def classify_trend(scores_0_100: list[float]) -> TrendTag:
        """Classify a series of per-screen scores into a trend tag.

        Rules (D-12):
          - < 2 points → flat
          - All significant deltas (|d| >= DELTA_FLOOR) positive → rising
          - All significant deltas negative → dropping
          - No deltas above floor → flat
          - >= VOLATILITY_FLIPS_MIN sign changes in significant deltas → volatile
          - Otherwise (mixed but few flips) → volatile (conservative default)
        """
        if len(scores_0_100) < 2:
            return "flat"
        deltas = [scores_0_100[i] - scores_0_100[i - 1] for i in range(1, len(scores_0_100))]
        significant = [d for d in deltas if abs(d) >= DELTA_FLOOR]
        if not significant:
            return "flat"
        flips = sum(
            1 for i in range(1, len(significant))
            if (significant[i] > 0) != (significant[i - 1] > 0)
        )
        if flips >= VOLATILITY_FLIPS_MIN:
            return "volatile"
        if all(d > 0 for d in significant):
            return "rising"
        if all(d < 0 for d in significant):
            return "dropping"
        return "volatile"


    def compute_flow_continuity(roi_per_screen: list[dict]) -> FlowContinuity:
        """Per-circuit per-screen score_0_100 + delta + trend over the full flow.

        Args:
            roi_per_screen: list of {"screen_index": int, "circuits": {name: {"score", "peak"}}, "regions": [...]}
                            Scores must already be rescaled to 0..100 (D-14).

        Returns FlowContinuity with one CircuitContinuity per named circuit.
        Missing circuit keys in any screen are treated as score=0.0 (should not happen
        with well-formed inputs from aggregate_screen_roi).
        """
        # Sort by screen_index to ensure consistent ordering
        sorted_screens = sorted(roi_per_screen, key=lambda s: s["screen_index"])

        circuits_out: dict[str, CircuitContinuity] = {}
        for circuit_name in CIRCUIT_NAMES:
            points: list[ContinuityPoint] = []
            prev_score: float | None = None
            for entry in sorted_screens:
                score = float(entry["circuits"].get(circuit_name, {}).get("score", 0.0))
                delta = 0.0 if prev_score is None else (score - prev_score)
                points.append(ContinuityPoint(
                    screen_index=int(entry["screen_index"]),
                    score_0_100=score,
                    delta_from_prev=delta,
                ))
                prev_score = score
            score_series = [p.score_0_100 for p in points]
            trend = classify_trend(score_series)
            circuits_out[circuit_name] = CircuitContinuity(per_screen=points, trend=trend)

        return FlowContinuity(circuits=circuits_out)
    ```
