3. Create `backend/ml/tribe_loader.py`:
```python
from tribev2 import TribeModel
import asyncio
import os

_model: TribeModel | None = None

async def load_model(cache_folder: str, device: str) -> TribeModel:
    """Load TRIBEv2 once at server startup. Blocks for 30-60s on first run (downloads weights).
    Subsequent runs use cached weights (~5-10s).
    Per D-08: caller must handle exceptions — if this fails, server refuses to start."""
    global _model
    loop = asyncio.get_event_loop()
    _model = await loop.run_in_executor(
        None,
        lambda: TribeModel.from_pretrained(
            "facebook/tribev2",
            cache_folder=cache_folder,
            device=device,  # "mps" or "cpu" — TRIBEv2 only auto-detects "cuda"
        ),
    )
    _model.eval()  # CRITICAL: BatchNorm/Dropout differ in train mode
    return _model

def get_model() -> TribeModel:
    """Get the loaded model instance. Raises RuntimeError if not loaded."""
    if _model is None:
        raise RuntimeError("TRIBEv2 model not loaded — was lifespan called?")
    return _model
```
