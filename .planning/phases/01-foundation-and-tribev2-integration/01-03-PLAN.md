```python
import tempfile
import os
from moviepy.editor import ImageClip, concatenate_videoclips

def create_slideshow(
    image_paths: list[str],
    durations: list[float],
    output_path: str | None = None,
    fps: int = 24,
) -> str:
    """Assemble screen images into a timed MP4 slideshow for TRIBEv2 video pathway.
    Per D-02: video slideshow path built alongside text in Phase 1.
    Per SIM-03: this is the fallback path if text differentiation fails.

    Args:
        image_paths: List of image file paths (PNG/JPG)
        durations: Duration in seconds for each image (must match len of image_paths)
        output_path: Where to write the MP4. If None, creates a temp file.
        fps: Frame rate for the output video.

    Returns:
        Path to the generated MP4 file.
    """
    if len(image_paths) != len(durations):
        raise ValueError(f"image_paths ({len(image_paths)}) and durations ({len(durations)}) must have same length")

    clips = []
    for img_path, dur in zip(image_paths, durations):
        clip = ImageClip(img_path, duration=dur)
        clips.append(clip)

    video = concatenate_videoclips(clips, method="compose")

    if output_path is None:
        tmp = tempfile.NamedTemporaryFile(suffix=".mp4", delete=False)
        output_path = tmp.name
        tmp.close()

    video.write_videofile(output_path, fps=fps, logger=None)  # logger=None suppresses moviepy output
    video.close()
    for clip in clips:
        clip.close()

    return output_path
```

```python
import numpy as np

# D-10: Consumer-friendly circuit names ONLY. No Brodmann codes, no anatomical jargon.
# D-11: Cortical proxies hardcoded to Destrieux atlas regions.
#        If Meta releases subcortical checkpoint (Issue #23), this is a code change.
#
# IMPORTANT: The exact Destrieux label strings MUST be verified against
# nilearn.datasets.fetch_atlas_destrieux_2009() at runtime. The labels below
# are the expected strings — verify_atlas_labels() confirms them.

ROI_CIRCUIT_MAP: dict[str, list[str]] = {
    "Attention": [
        "S_front_inf",
        "G_front_inf-Triangul",
        "S_intrapariet_and_P_trans",
    ],
    "Cognitive Load": [
        "G_front_middle",
        "S_front_middle",
        "G_front_sup",
    ],
    "Emotional Response": [
        "G_insular_short",
        "S_cingul-Marginalis",
        "G_cingul-Post-ventral",
    ],
    "Memory Encoding": [
        "G_oc-temp_med-Parahip",
        "S_collat_transv_ant",
    ],
    "Reward": [
        "G_orbital",
        "G_rectus",
        "S_orbital-H_Shaped",
    ],
}

# All five circuit names — used for validation
CIRCUIT_NAMES = list(ROI_CIRCUIT_MAP.keys())

# Subcortical labels that must NEVER appear anywhere in output
SUBCORTICAL_LABELS = [
    "striatum", "putamen", "caudate", "nucleus accumbens",
    "hippocampus", "amygdala", "thalamus", "hypothalamus",
    "substantia nigra", "ventral tegmental",
]


def get_atlas_labels() -> list[str]:
    """Fetch Destrieux atlas label names from nilearn.
    Returns list of region label strings indexed by vertex region assignment."""
    from nilearn.datasets import fetch_atlas_destrieux_2009
    atlas = fetch_atlas_destrieux_2009()
    # atlas.labels is a list of label strings
    return [label.decode("utf-8") if isinstance(label, bytes) else str(label) for label in atlas.labels]


def verify_atlas_labels() -> dict[str, list[str]]:
    """Verify that all ROI_CIRCUIT_MAP region names exist in the Destrieux atlas.
    Returns dict of circuit_name -> list of missing region names (empty if all found)."""
    atlas_labels = get_atlas_labels()
    missing = {}
    for circuit, regions in ROI_CIRCUIT_MAP.items():
        not_found = [r for r in regions if r not in atlas_labels]
        if not_found:
            missing[circuit] = not_found
    return missing


def aggregate_roi_scores(
    mean_activation: np.ndarray,
    atlas_labels: list[str],
) -> dict[str, dict[str, float]]:
    """Map per-vertex mean activation to five named circuits.
    Per D-10: returns consumer-friendly names only.
    Per D-11: cortical proxies only — no subcortical.

    Args:
        mean_activation: 1D array of shape (~20k,) — mean activation per vertex
        atlas_labels: List of region label strings, one per vertex (from Destrieux atlas)

    Returns:
        Dict mapping circuit name to {"score": float, "peak": float}
        Example: {"Attention": {"score": 0.42, "peak": 0.78}, ...}
    """
    results = {}
    for circuit_name, region_names in ROI_CIRCUIT_MAP.items():
        indices = [i for i, label in enumerate(atlas_labels) if label in region_names]
        if indices:
            scores = mean_activation[indices]
            results[circuit_name] = {
                "score": float(np.mean(scores)),
                "peak": float(np.max(scores)),
            }
        else:
            results[circuit_name] = {"score": 0.0, "peak": 0.0}
    return results
```
