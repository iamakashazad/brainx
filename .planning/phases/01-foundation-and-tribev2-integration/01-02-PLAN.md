1. Create `backend/services/inference.py` — TRIBEv2 wrapper with text and video path support:
```python
import asyncio
import torch
import numpy as np
import tempfile
import os
from concurrent.futures import ThreadPoolExecutor
from backend.ml.tribe_loader import get_model

gpu_executor = ThreadPoolExecutor(max_workers=1)

async def run_tribev2_text(text_content: str) -> np.ndarray:
    """Run TRIBEv2 inference via text pathway (gTTS internally).
    Writes text to a temp file, passes as text_path to TRIBEv2.
    Returns: np.ndarray of shape (n_timesteps, ~20k_vertices)"""
    loop = asyncio.get_event_loop()
    model = get_model()

    # Write text to temp file — TRIBEv2 expects a file path
    tmp = tempfile.NamedTemporaryFile(mode="w", suffix=".txt", delete=False)
    tmp.write(text_content)
    tmp.close()

    def _infer():
        try:
            with torch.inference_mode():
                df = model.get_events_dataframe(text_path=tmp.name)
                preds, segments = model.predict(events=df)
            return preds.cpu().numpy()  # move off GPU before returning
        finally:
            os.unlink(tmp.name)

    return await loop.run_in_executor(gpu_executor, _infer)

async def run_tribev2_video(video_path: str) -> np.ndarray:
    """Run TRIBEv2 inference via video pathway.
    Expects an MP4 file path. Returns: np.ndarray of shape (n_timesteps, ~20k_vertices)"""
    loop = asyncio.get_event_loop()
    model = get_model()

    def _infer():
        with torch.inference_mode():
            df = model.get_events_dataframe(video_path=video_path)
            preds, segments = model.predict(events=df)
        return preds.cpu().numpy()

    return await loop.run_in_executor(gpu_executor, _infer)
```

2. Create `backend/core/queue.py` — asyncio.Queue with single GPU worker, retry logic per D-07:
```python
import asyncio
import numpy as np
import os
import traceback
from backend.core.database import async_session
from backend.models.job import Job
from backend.services.inference import run_tribev2_text, run_tribev2_video
from sqlalchemy import select

gpu_queue: asyncio.Queue = asyncio.Queue()

async def gpu_worker_loop():
    """Single GPU worker coroutine. Consumes jobs from gpu_queue one at a time.
    Per D-07: retries once on GPU failure, then marks ERROR."""
    while True:
        job_id = await gpu_queue.get()
        try:
            await _process_job(job_id)
        except Exception:
            # Unexpected error — should be caught inside _process_job
            pass
        finally:
            gpu_queue.task_done()

async def _process_job(job_id: str):
    """Process a single inference job with retry logic."""
    async with async_session() as session:
        result = await session.execute(select(Job).where(Job.id == job_id))
        job = result.scalar_one_or_none()
        if job is None:
            return

        # Transition to RUNNING
        job.status = "RUNNING"
        job.stage = "inference"
        job.queue_position = None
        await session.commit()

    # Attempt inference (with one retry per D-07)
    max_retries = 1
    last_error = None

    for attempt in range(max_retries + 1):
        try:
            async with async_session() as session:
                result = await session.execute(select(Job).where(Job.id == job_id))
                job = result.scalar_one()

                if job.input_type == "video" and job.input_path:
                    activation = await run_tribev2_video(job.input_path)
                else:
                    # Default: text path — input_path contains the text content for Phase 1
                    activation = await run_tribev2_text(job.input_path or "")

                # Save activation array to disk (not in SQLite — too large)
                os.makedirs("activations", exist_ok=True)
                npy_path = f"activations/{job_id}.npy"
                np.save(npy_path, activation)

                # Mark DONE
                job.status = "DONE"
                job.stage = None
                job.result_json = {
                    "activation_path": npy_path,
                    "shape": list(activation.shape),
                }
                job.retry_count = attempt
                await session.commit()
                return  # Success

        except Exception as e:
            last_error = str(e)
            if attempt < max_retries:
                continue  # retry once
            # Final failure — mark ERROR
            async with async_session() as session:
                result = await session.execute(select(Job).where(Job.id == job_id))
                job = result.scalar_one()
                job.status = "ERROR"
                job.stage = None
                job.error_message = f"GPU inference failed after {attempt + 1} attempt(s): {last_error}"
                job.retry_count = attempt + 1
                await session.commit()
```

```python
from fastapi import APIRouter, HTTPException
from backend.models.schemas import TestInferenceRequest, TestInferenceResponse, JobStatusResponse
from backend.models.job import Job
from backend.core.database import async_session
from backend.core.queue import gpu_queue
from sqlalchemy import select
import uuid

router = APIRouter()

@router.post("/test-inference", response_model=TestInferenceResponse)
async def submit_test_inference(request: TestInferenceRequest):
    """Submit a test inference job. Per D-05: test-only endpoint for Phase 1."""
    job_id = str(uuid.uuid4())
    async with async_session() as session:
        job = Job(
            id=job_id,
            status="PENDING",
            queue_position=gpu_queue.qsize() + 1,
            input_path=request.text,     # For Phase 1, store raw text in input_path
            input_type=request.input_type,
        )
        session.add(job)
        await session.commit()

    await gpu_queue.put(job_id)
    return TestInferenceResponse(
        job_id=job_id,
        status="PENDING",
        message=f"Job queued at position {gpu_queue.qsize()}",
    )

@router.get("/jobs/{job_id}", response_model=JobStatusResponse)
async def get_job_status(job_id: str):
    """Poll job status. Per D-05/INFRA-03: frontend tracks queue position without websockets."""
    async with async_session() as session:
        result = await session.execute(select(Job).where(Job.id == job_id))
        job = result.scalar_one_or_none()
        if job is None:
            raise HTTPException(status_code=404, detail=f"Job {job_id} not found")
        return JobStatusResponse(
            job_id=job.id,
            status=job.status,
            queue_position=job.queue_position,
            stage=job.stage,
            result=job.result_json,
            error=job.error_message,
            created_at=job.created_at.isoformat() if job.created_at else None,
        )
```

5. Create `backend/main.py` — FastAPI app with lifespan per D-08:
```python
from contextlib import asynccontextmanager
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from backend.ml.tribe_loader import load_model
from backend.core.queue import gpu_worker_loop
from backend.core.database import init_db
from backend.core.config import settings
from backend.routers import jobs, health
import asyncio
import sys

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Initialize database tables
    await init_db()

    # Load TRIBEv2 — per D-08: if this fails, server refuses to start
    try:
        app.state.tribe_model = await load_model(
            settings.model_cache_path, settings.device
        )
    except Exception as e:
        print(f"FATAL: TRIBEv2 failed to load: {e}", file=sys.stderr)
        sys.exit(1)

    # Start the single GPU worker coroutine
    worker_task = asyncio.create_task(gpu_worker_loop())

    yield

    # Shutdown: cancel worker
    worker_task.cancel()

app = FastAPI(title="BrainX", version="0.1.0", lifespan=lifespan)

# CORS — allow local frontend dev server
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173", "http://localhost:3000"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(health.router, tags=["health"])
app.include_router(jobs.router, tags=["jobs"])
```
